// Pre-money pool shuffle (investor demands pre-money pool)
totalAfterESOP = founderShares / (1 - esopPct/100)
esopShares     = totalAfterESOP - founderShares
// Investor then buys into the inflated share count
investorShares = totalAfterESOP × invPct / (1 - invPct)

// Post-money pool (pool comes from everyone post-close)
totalPostRound = founderShares + investorShares
esopShares     = totalPostRound × esopPct / (1 - esopPct)
// Founders diluted only by their proportionate share
